# K-means Clustering Function
def k_means(X, K, max_iters=100, tol=1e-4):
    np.random.seed(42)
    centers = X[np.random.choice(X.shape[0], K, replace=False)]
    for i in range(max_iters):
        distances = np.sqrt(((X - centers[:, np.newaxis]) ** 2).sum(axis=2))
        labels = distances.argmin(axis=0)
        new_centers = np.array([X[labels == k].mean(axis=0) for k in range(K)])
        if np.all(np.abs(new_centers - centers) < tol):
            break
        centers = new_centers
    return centers, labels

# Correct Gaussian Basis Function with scale factor S
def gaussian_basis(X, centers, S):
    # Compute Gaussian basis functions with the specified scale factor S
    return np.exp(-cdist(X, centers)**2 / S)

for M in complexities:
    # Perform K-means clustering with K = M
    centers, _ = k_means(X_train, M)
    
    # Calculate the scale factor S using the formula
    max_distance = np.max(cdist(centers, centers))
    S = max_distance / np.sqrt(2 * M)
    
    # Compute Gaussian basis functions for training and test data
    Phi_train = gaussian_basis(X_train, centers, S)
    Phi_test = gaussian_basis(X_test, centers, S)
    
    # Regularization parameters for cross-validation
    regularization_params = [0, 0.001, 0.01, 0.1, 1, 10]
    kf = KFold(n_splits=5, shuffle=True, random_state=42)
    
    train_errors, test_errors = [], []
    
    for reg in regularization_params:
        fold_train_errors, fold_test_errors = [], []
        
        for train_idx, val_idx in kf.split(Phi_train):
            X_train_fold, X_val_fold = Phi_train[train_idx], Phi_train[val_idx]
            y_train_fold, y_val_fold = y_train[train_idx], y_train[val_idx]
            
            # Compute weights with regularization
            w = np.linalg.inv(X_train_fold.T @ X_train_fold + reg * np.eye(X_train_fold.shape[1])) @ X_train_fold.T @ y_train_fold
            
            # Training and validation predictions
            fold_train_errors.append(mean_squared_error(y_train_fold, X_train_fold @ w))
            fold_test_errors.append(mean_squared_error(y_val_fold, X_val_fold @ w))
        
        # Record cross-validated errors
        train_errors.append(np.mean(fold_train_errors))
        test_errors.append(np.mean(fold_test_errors))
    
    # Select best regularization
    best_reg = regularization_params[np.argmin(test_errors)]
    w_best = np.linalg.inv(Phi_train.T @ Phi_train + best_reg * np.eye(Phi_train.shape[1])) @ Phi_train.T @ y_train
    
    # Model predictions and MSE for train and test data
    mse_train_results[M] = mean_squared_error(y_train, Phi_train @ w_best)
    mse_test_results[M] = mean_squared_error(y_test, Phi_test @ w_best)
    
    # Plot Model Output vs Target Output (Approximated Function Plot)
    plt.figure()
    plt.plot(y_train, Phi_train @ w_best, 'bo', label='Train Prediction')
    plt.plot(y_test, Phi_test @ w_best, 'ro', label='Test Prediction')
    plt.xlabel("Target Output")
    plt.ylabel("Model Output")
    plt.title(f"Model Complexity: {M}, Best Regularization: {best_reg}")
    plt.legend()
    plt.show()

# Plot Mean Squared Error for Different Model Complexities
